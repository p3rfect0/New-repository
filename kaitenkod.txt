#include <iostream>
#include <string>
#include <limits>
#include <algorithm>

using namespace std;

struct Time {
    int hours;
    int minutes;
    int seconds;
};

// Функция ввода времени с проверкой корректности
Time inputTime() {
    Time t;

    do {
        cout << "Часы (0-23): ";
        cin >> t.hours;
        if (t.hours < 0 || t.hours > 23) {
            cout << "Ошибка! Часы должны быть в диапазоне 0-23. " << endl;
        }
    } while (t.hours < 0 || t.hours > 23);

    do {
        cout << "Минуты (0-59): ";
        cin >> t.minutes;
        if (t.minutes < 0 || t.minutes > 59) {
            cout << "Ошибка! Минуты должны быть в диапазоне 0-59. " << endl;
        }
    } while (t.minutes < 0 || t.minutes > 59);

    do {
        cout << "Секунды (0-59): ";
        cin >> t.seconds;
        if (t.seconds < 0 || t.seconds > 59) {
            cout << "Ошибка! Секунды должны быть в диапазоне 0-59. " << endl;
        }
    } while (t.seconds < 0 || t.seconds > 59);

    return t;
}

// Функция для вывода времени
void printTime(const Time& t) {
    cout << (t.hours < 10 ? "0" : "") << t.hours << ":"
        << (t.minutes < 10 ? "0" : "") << t.minutes << ":"
        << (t.seconds < 10 ? "0" : "") << t.seconds;
}

// Функция для сравнения двух времен (возвращает true, если t1 > t2)
bool isTimeGreater(const Time& t1, const Time& t2) {
    if (t1.hours != t2.hours) return t1.hours > t2.hours;
    if (t1.minutes != t2.minutes) return t1.minutes > t2.minutes;
    return t1.seconds > t2.seconds;
}

// Функция для сравнения двух времен (возвращает true, если t1 < t2)
bool isTimeLess(const Time& t1, const Time& t2) {
    if (t1.hours != t2.hours) return t1.hours < t2.hours;
    if (t1.minutes != t2.minutes) return t1.minutes < t2.minutes;
    return t1.seconds < t2.seconds;
}


// Функция для проверки равенства двух времен
bool isTimeEqual(const Time& t1, const Time& t2) {
    return t1.hours == t2.hours && t1.minutes == t2.minutes && t1.seconds == t2.seconds;
}

// Прямой (последовательный) поиск
int linearSearch(Time* arr, int size, const Time& target, int& comparisons) {
    comparisons = 0;
    for (int i = 0; i < size; i++) {
        comparisons++;
        if (isTimeEqual(arr[i], target)) {
            return i; // Найден элемент, возвращаем индекс
        }
    }
    return -1; // Элемент не найден
}

// Бинарный поиск (сортировка массива по возрастанию)
int binarySearch(Time* arr, int size, const Time& target, int& comparisons) {
    comparisons = 0;
    int left = 0;
    int right = size - 1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        comparisons++;

        if (isTimeEqual(arr[mid], target)) {
            return mid; // Найден элемент
        }
        else if (isTimeLess(arr[mid], target)) {
            comparisons++;
            left = mid + 1;

        }
        else {
            comparisons++;
            right = mid - 1;
        }
    }
    return -1;
}

// Сортировка обменом (пузырьковая сортировка)
void bubbleSort(Time* arr, int size, int& comparisons, int& swaps) {
    comparisons = 0;
    swaps = 0;

    for (int i = 0; i < size - 1; i++) {
        for (int j = 0; j < size - i - 1; j++) {
            comparisons++;
            if (isTimeGreater(arr[j], arr[j + 1])) {
                swap(arr[j], arr[j + 1]);
                swaps++;
            }
        }
    }
}

// Сортировка вставками
void insertionSort(Time* arr, int size, int& comparisons, int& swaps) {
    comparisons = 0;
    swaps = 0;

    for (int i = 1; i < size; i++) {
        Time key = arr[i];
        int j = i - 1;

        comparisons++;
        while (j >= 0 && isTimeGreater(arr[j], key)) {
            comparisons++;
            arr[j + 1] = arr[j];
            swaps++;
            j--;
        }
        arr[j + 1] = key;
        if (j + 1 != i) swaps++;
    }
}

// Сортировка Шелла
void shellSort(Time* arr, int size, int& comparisons, int& swaps) {
    comparisons = 0;
    swaps = 0;

    // Последовательность gaps (можно оптимизировать)
    for (int gap = size / 2; gap > 0; gap /= 2) {
        for (int i = gap; i < size; i++) {
            Time temp = arr[i];
            int j;

            comparisons++;
            for (j = i; j >= gap && isTimeGreater(arr[j - gap], temp); j -= gap) {
                comparisons++;
                arr[j] = arr[j - gap];
                swaps++;
            }
            arr[j] = temp;
            if (j != i) swaps++;
        }
    }
}

// Вспомогательная функция для пирамидальной сортировки (просеивание вниз)
void heapify(Time* arr, int n, int i, int& comparisons, int& swaps) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n) {
        comparisons++;
        if (isTimeGreater(arr[left], arr[largest])) {
            largest = left;
        }
    }

    if (right < n) {
        comparisons++;
        if (isTimeGreater(arr[right], arr[largest])) {
            largest = right;
        }
    }

    if (largest != i) {
        swap(arr[i], arr[largest]);
        swaps++;
        heapify(arr, n, largest, comparisons, swaps);
    }
}

// Пирамидальная сортировка (Heap Sort)
void heapSort(Time* arr, int size, int& comparisons, int& swaps) {
    comparisons = 0;
    swaps = 0;

    // Построение max-heap
    for (int i = size / 2 - 1; i >= 0; i--) {
        heapify(arr, size, i, comparisons, swaps);
    }

    // Извлечение элементов из кучи
    for (int i = size - 1; i > 0; i--) {
        swap(arr[0], arr[i]);
        swaps++;
        heapify(arr, i, 0, comparisons, swaps);
    }
}

// Функция для вывода массива времен
void printTimeArray(Time* arr, int size) {
    for (int i = 0; i < size; i++) {
        cout << i + 1 << ". ";
        printTime(arr[i]);
        cout << endl;
    }
}

                printTimeArray(sortedArray, arraySize);
